// Simple hash function (for demo; in production use bcrypt server-side)
async function hashPassword(pw) {
  const buf = await crypto.subtle.digest("SHA-256", new TextEncoder().encode(pw));
  return Array.from(new Uint8Array(buf)).map(b => b.toString(16).padStart(2, "0")).join("");
}

// ─── STORAGE HELPERS ──────────────────────────────────────────
async function storageGet(key) {
  try { const r = await window.storage.get(key); return r ? JSON.parse(r.value) : null; } catch { return null; }
}
async function storageSet(key, value) {
  try { await window.storage.set(key, JSON.stringify(value)); return true; } catch { return false; }
}
async function storageList(prefix) {
  try { const r = await window.storage.list(prefix); return r ? r.keys : []; } catch { return []; }
}
async function storageDelete(key) {
  try { await window.storage.delete(key); return true; } catch { return false; }
}

// ─── SESSION ──────────────────────────────────────────────────
function getSession() {
  try {
    const s = JSON.parse(sessionStorage.getItem(SESSION_KEY) || "null");
    if (s && Date.now() - s.ts < SESSION_DURATION) return s;
    sessionStorage.removeItem(SESSION_KEY);
    return null;
  } catch { return null; }
}
function setSession() {
  sessionStorage.setItem(SESSION_KEY, JSON.stringify({ ts: Date.now() }));
}
function clearSession() {
  sessionStorage.removeItem(SESSION_KEY);
}

// ─── LOGIN PAGE ───────────────────────────────────────────────
function LoginPage({ onLogin }) {
  const [pw, setPw] = useState("");
  const [err, setErr] = useState("");
  const [loading, setLoading] = useState(false);
  const [attempts, setAttempts] = useState(0);
  const [locked, setLocked] = useState(false);

  useEffect(() => {
    const link = document.createElement("link");
    link.rel = "preconnect"; link.href = "https://fonts.googleapis.com";
    document.head.appendChild(link);
    const link2 = document.createElement("link");
    link2.rel = "stylesheet";
    link2.href = "https://fonts.googleapis.com/css2?family=Syne:wght@700;800&family=DM+Sans:wght@300;400;500&display=swap";
    document.head.appendChild(link2);
  }, []);

  async function handleLogin(e) {
    e.preventDefault();
    if (locked) return;
    if (!pw) { setErr("Please enter your password."); return; }
    setLoading(true); setErr("");
    await new Promise(r => setTimeout(r, 400)); // slight delay for UX
    const hash = await hashPassword(pw);
    // Load stored hash or use default
    let storedHash = await storageGet(ADMIN_PASSWORD_HASH_KEY);
    if (!storedHash) {
      storedHash = await hashPassword(DEFAULT_PASSWORD);
      await storageSet(ADMIN_PASSWORD_HASH_KEY, storedHash);
    }
    if (hash === storedHash) {
      setSession();
      onLogin();
    } else {
      const newAttempts = attempts + 1;
      setAttempts(newAttempts);
      if (newAttempts >= 5) {
        setLocked(true);
        setErr("Too many failed attempts. Session locked for 5 minutes.");
        setTimeout(() => { setLocked(false); setAttempts(0); }, 5 * 60 * 1000);
      } else {
        setErr(`Incorrect password. ${5 - newAttempts} attempt(s) remaining.`);
      }
    }
    setLoading(false);
  }

  return (
    <div style={S.loginWrap}>
      <div style={S.loginCard}>
        <div style={S.logo}>Delivery<span style={S.logoSpan}>Zone</span></div>
        <div style={{ fontFamily: "'Syne', sans-serif", fontWeight: 700, color: "#8BA4BC", fontSize: "0.85rem", marginBottom: "0.3rem" }}>Admin Dashboard</div>
        <div style={S.loginSub}>Sign in to manage products and snapshots</div>

        {err && <div style={S.error}>🔒 {err}</div>}

        <form onSubmit={handleLogin}>
          <label style={{ ...S.label, marginTop: "1rem" }}>Password</label>
          <input
            type="password"
            style={{ ...S.input, marginBottom: "1.5rem" }}
            value={pw}
            onChange={e => setPw(e.target.value)}
            placeholder="Enter admin password"
            autoFocus
            disabled={locked}
          />
          <button type="submit" style={{ ...S.btn, opacity: loading || locked ? 0.7 : 1 }} disabled={loading || locked}>
            {loading ? "Verifying…" : locked ? "Locked" : "Sign In →"}
          </button>
        </form>

        <div style={{ marginTop: "1.5rem", padding: "1rem", background: "rgba(245,158,11,0.06)", borderRadius: 10, border: "1px solid rgba(245,158,11,0.12)" }}>
          <div style={{ fontSize: "0.75rem", color: "#8BA4BC", lineHeight: 1.7 }}>
            <strong style={{ color: "#F59E0B" }}>🔐 Security</strong><br/>
            Default password: <code style={{ color: "#F0F6FF" }}>DeliveryZone2025!</code><br/>
            Change it in Settings after login.
          </div>
        </div>
      </div>
    </div>
  );
}

// ─── PRODUCT FORM ─────────────────────────────────────────────
function ProductForm({ initial, onSave, onCancel }) {
  const blank = { name: "", category: "", price: "", originalPrice: "", rating: "4.5", image: "", link: "", description: "", featured: false };
  const [f, setF] = useState(initial || blank);
  const [err, setErr] = useState("");

  function set(k, v) { setF(p => ({ ...p, [k]: v })); }

  function submit() {
    if (!f.name || !f.category || !f.price || !f.link) { setErr("Name, Category, Price and Affiliate Link are required."); return; }
    onSave({ ...f, id: f.id || Date.now(), rating: parseFloat(f.rating) || 4.0 });
  }

  return (
    <div>
      {err && <div style={S.error}>{err}</div>}
      <div style={S.formGrid}>
        {[["Product Name *", "name", "text", "Sony WH-1000XM5…"], ["Category *", "category", "text", "Electronics, Fashion…"],
          ["Price *", "price", "text", "$199.99"], ["Original Price", "originalPrice", "text", "$349.99 (optional)"],
          ["Rating (1–5)", "rating", "number", "4.5"], ["Image URL", "image", "url", "https://…"]].map(([lbl, key, type, ph]) => (
          <div key={key} style={S.formField}>
            <label style={S.label}>{lbl}</label>
            <input style={S.input} type={type} placeholder={ph} value={f[key]}
              onChange={e => set(key, e.target.value)} min={type === "number" ? 1 : undefined} max={type === "number" ? 5 : undefined} step={type === "number" ? "0.1" : undefined} />
          </div>
        ))}
      </div>
      <div style={{ ...S.formField, marginTop: "1rem" }}>
        <label style={S.label}>Affiliate Link *</label>
        <input style={S.input} type="url" placeholder="https://amzn.to/…" value={f.link} onChange={e => set("link", e.target.value)} />
      </div>
      <div style={{ ...S.formField, marginTop: "1rem" }}>
        <label style={S.label}>Description</label>
        <textarea style={S.textarea} placeholder="Why you'll love this product…" value={f.description} onChange={e => set("description", e.target.value)} />
      </div>
      <label style={{ display: "flex", alignItems: "center", gap: "0.6rem", marginTop: "1rem", cursor: "pointer", fontSize: "0.88rem", color: "#8BA4BC" }}>
        <input type="checkbox" checked={f.featured} onChange={e => set("featured", e.target.checked)} /> Mark as Featured Product
      </label>
      <div style={{ display: "flex", gap: "0.8rem", marginTop: "1.5rem" }}>
        <button style={S.btn} onClick={submit}>{initial ? "Save Changes" : "Add Product"}</button>
        <button style={{ ...S.btnOutline, padding: "0.8rem 1.5rem" }} onClick={onCancel}>Cancel</button>
      </div>
    </div>
  );
}

// ─── MAIN ADMIN APP ───────────────────────────────────────────
function AdminApp({ onLogout }) {
  const [page, setPage] = useState("dashboard");
  const [products, setProducts] = useState([]);
  const [snapshots, setSnapshots] = useState([]); // [{date, key}]
  const [loading, setLoading] = useState(true);
  const [toast, setToast] = useState("");
  const [modal, setModal] = useState(null); // {type, data}
  const [newPw, setNewPw] = useState(""); const [newPw2, setNewPw2] = useState("");
  const [pwMsg, setPwMsg] = useState(null);
  const [genPreview, setGenPreview] = useState(null);
  const [snapshotFilter, setSnapshotFilter] = useState("");

  function showToast(msg) { setToast(msg); setTimeout(() => setToast(""), 3000); }

  // Load current products and snapshot list
  useEffect(() => { load(); }, []);

  async function load() {
    setLoading(true);
    const prods = await storageGet("dz:current_products");
    setProducts(prods || SAMPLE);
    if (!prods) await storageSet("dz:current_products", SAMPLE);

    const keys = await storageList("dz:snap:");
    const snaps = keys.map(k => ({ key: k, date: k.replace("dz:snap:", "") })).sort((a, b) => b.date.localeCompare(a.date));
    setSnapshots(snaps);
    setLoading(false);
  }

  // ── PRODUCT OPS ─────────────────────────────────────────────
  async function saveProducts(updated) {
    setProducts(updated);
    await storageSet("dz:current_products", updated);
  }

  async function addProduct(p) {
    const updated = [p, ...products];
    await saveProducts(updated);
    setModal(null);
    showToast("✅ Product added!");
  }

  async function editProduct(p) {
    const updated = products.map(x => x.id === p.id ? p : x);
    await saveProducts(updated);
    setModal(null);
    showToast("✅ Product updated!");
  }

  async function deleteProduct(id) {
    if (!confirm("Delete this product?")) return;
    const updated = products.filter(p => p.id !== id);
    await saveProducts(updated);
    showToast("🗑️ Product deleted.");
  }

  // ── SNAPSHOTS ────────────────────────────────────────────────
  async function saveSnapshot() {
    const date = todayStr();
    const key = `dz:snap:${date}`;
    const snap = { date, products: [...products], createdAt: new Date().toISOString() };
    await storageSet(key, snap);
    showToast(`📸 Snapshot saved for ${fmtDate(date)}`);
    await load();
  }

  async function deleteSnapshot(key) {
    if (!confirm("Delete this snapshot?")) return;
    await storageDelete(key);
    showToast("🗑️ Snapshot deleted.");
    await load();
  }

  async function generatePage(snap) {
    const html = generateHTML(snap);
    setGenPreview({ html, date: snap.date, snap });
  }

  async function generateFromKey(key) {
    const snap = await storageGet(key);
    if (snap) generatePage(snap);
  }

  function downloadHTML(html, date) {
    const blob = new Blob([html], { type: "text/html" });
    const a = document.createElement("a");
    a.href = URL.createObjectURL(blob);
    a.download = `deliveryzone-${date}.html`;
    a.click();
  }

  // ── CHANGE PASSWORD ──────────────────────────────────────────
  async function changePassword() {
    if (!newPw || newPw.length < 8) { setPwMsg({ type: "error", msg: "Password must be at least 8 characters." }); return; }
    if (newPw !== newPw2) { setPwMsg({ type: "error", msg: "Passwords do not match." }); return; }
    const hash = await hashPassword(newPw);
    await storageSet(ADMIN_PASSWORD_HASH_KEY, hash);
    setPwMsg({ type: "success", msg: "✅ Password updated!" });
    setNewPw(""); setNewPw2("");
  }

  // ── NAV ──────────────────────────────────────────────────────
  const nav = [
    { id: "dashboard", icon: "◈", label: "Dashboard" },
    { id: "products", icon: "⊞", label: "Products" },
    { id: "snapshots", icon: "📅", label: "Snapshots" },
    { id: "generate", icon: "⚡", label: "Generate Page" },
    { id: "settings", icon: "⚙", label: "Settings" },
  ];

  const stats = [
    { label: "Total Products", val: products.length, icon: "⊞" },
    { label: "Categories", val: [...new Set(products.map(p => p.category))].length, icon: "🏷" },
    { label: "Snapshots", val: snapshots.length, icon: "📅" },
    { label: "Featured", val: products.filter(p => p.featured).length, icon: "⭐" },
  ];

  if (loading) return <div style={{ ...S.app, display: "flex", alignItems: "center", justifyContent: "center", minHeight: "100vh" }}><div style={{ color: "#F59E0B", fontFamily: "'Syne', sans-serif", fontSize: "1.2rem" }}>Loading DeliveryZone…</div></div>;

  return (
    <div style={S.app}>
      {/* TOAST */}
      {toast && <div style={{ position: "fixed", top: "1.5rem", right: "1.5rem", background: "#1A2F44", border: "1px solid rgba(245,158,11,0.3)", color: "#F0F6FF", padding: "0.8rem 1.3rem", borderRadius: 12, zIndex: 600, fontSize: "0.9rem", boxShadow: "0 4px 20px rgba(0,0,0,0.4)", animation: "none" }}>{toast}</div>}

      {/* MODAL */}
      {modal && (
        <div style={S.overlay} onClick={e => e.target === e.currentTarget && setModal(null)}>
          <div style={S.modal}>
            <div style={S.modalHeader}>
              <span style={{ fontFamily: "'Syne', sans-serif", fontWeight: 700, fontSize: "1.1rem" }}>{modal.type === "add" ? "Add Product" : "Edit Product"}</span>
              <button onClick={() => setModal(null)} style={{ background: "none", border: "none", color: "#8BA4BC", cursor: "pointer", fontSize: "1.1rem" }}>✕</button>
            </div>
            <div style={S.modalBody}>
              <ProductForm initial={modal.data} onSave={modal.type === "add" ? addProduct : editProduct} onCancel={() => setModal(null)} />
            </div>
          </div>
        </div>
      )}

      {/* PREVIEW MODAL */}
      {genPreview && (
        <div style={S.overlay}>
          <div style={{ ...S.modal, maxWidth: 780, display: "flex", flexDirection: "column" }}>
            <div style={S.modalHeader}>
              <span style={{ fontFamily: "'Syne', sans-serif", fontWeight: 700, fontSize: "1.1rem" }}>🖥 Preview — {fmtDate(genPreview.date)}</span>
              <div style={{ display: "flex", gap: "0.6rem", alignItems: "center" }}>
                <button style={S.btnSm} onClick={() => downloadHTML(genPreview.html, genPreview.date)}>⬇ Download HTML</button>
                <button onClick={() => setGenPreview(null)} style={{ background: "none", border: "none", color: "#8BA4BC", cursor: "pointer", fontSize: "1.1rem" }}>✕</button>
              </div>
            </div>
            <div style={{ padding: "1rem 1.5rem 0.5rem", color: "#8BA4BC", fontSize: "0.82rem" }}>Live preview below — download the HTML file to deploy.</div>
            <iframe
              srcDoc={genPreview.html}
              style={{ flex: 1, height: 500, border: "none", borderRadius: "0 0 18px 18px" }}
              title="Preview"
            />
          </div>
        </div>
      )}

      <div style={S.shell}>
        {/* SIDEBAR */}
        <aside style={S.sidebar}>
          <div style={S.sidebarLogo}>Delivery<span style={{ color: "#F59E0B" }}>Zone</span></div>
          {nav.map(n => (
            <button key={n.id} style={{ ...S.navItem, ...(page === n.id ? S.navItemActive : {}) }} onClick={() => setPage(n.id)}>
              <span>{n.icon}</span>{n.label}
            </button>
          ))}
          <div style={{ flex: 1 }} />
          <button style={{ ...S.navItem, color: "#ef4444" }} onClick={() => { clearSession(); onLogout(); }}>
            <span>⎋</span> Sign Out
          </button>
        </aside>

        {/* MAIN */}
        <main style={S.main}>

          {/* ── DASHBOARD ── */}
          {page === "dashboard" && (
            <div>
              <div style={S.pageTitle}>Dashboard</div>
              <div style={S.pageSub}>Welcome back — here's your DeliveryZone overview.</div>
              <div style={{ display: "grid", gridTemplateColumns: "repeat(auto-fill, minmax(200px, 1fr))", gap: "1rem", marginBottom: "2rem" }}>
                {stats.map(s => (
                  <div key={s.label} style={{ ...S.card, marginBottom: 0 }}>
                    <div style={{ fontSize: "1.8rem", marginBottom: "0.5rem" }}>{s.icon}</div>
                    <div style={{ fontFamily: "'Syne', sans-serif", fontWeight: 800, fontSize: "2rem", color: "#F59E0B" }}>{s.val}</div>

                  • No data is shared with other users

                  }}>🗑 Delete All Snapshots</button>
                </div>
              </div>
            </div>
          )}

        </main>
      </div>
    </div>
  );
}

// ─── ROOT ─────────────────────────────────────────────────────
export default function App() {
  const [authed, setAuthed] = useState(!!getSession());

  // Refresh session on activity
  useEffect(() => {
    if (!authed) return;
    const refresh = () => setSession();
    window.addEventListener("click", refresh);
    window.addEventListener("keydown", refresh);
    return () => { window.removeEventListener("click", refresh); window.removeEventListener("keydown", refresh); };
  }, [authed]);

  if (!authed) return <LoginPage onLogin={() => setAuthed(true)} />;
  return <AdminApp onLogout={() => setAuthed(false)} />;
}
